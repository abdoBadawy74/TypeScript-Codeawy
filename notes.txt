**keyof**
interface IUser {
  username: string;
  age: number;
  address: string;
}

type UserKeys = keyof IUser; // * username | age | address

const user: IUser = {
  username: "codeeav",
  age: 30,
  address: "10 main street",
};

function getProperty(obj: IUser, key: UserKeys) {
  return obj[key];
}

console.log(getProperty(user, "username"));
console.log(getProperty(user, "age"));
console.log(getProperty(user, "address"));

enum StatusCode {
  Success = 200,
  NotFound = 404,
  ServerError = 500,
}

function handleResponse(status: StatusCode) {
  switch (status) {
    case StatusCode.Success:
      console.log("Success: The request was successful.");
      break;

    case StatusCode.NotFound:
      console.log("Not Found: The request was not found.");
      break;

    case StatusCode.ServerError:
      console.log("Server Error: The request errored on the server.");
      break;
    default:
       console.log("UnKnown status code : Unable to handle the response");
  }
}
handleResponse(StatusCode.Success);
